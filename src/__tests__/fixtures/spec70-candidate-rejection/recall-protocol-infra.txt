const API = "https://api.hetzner.cloud/v1";
const SCRIPT_DIR = dirname(resolve(process.argv[1] || "infra.ts"));

type HetznerServer = {
  id: number;
  name: string;
  status: string;
  created: string;
  public_net: { ipv4: { ip: string } };
  server_type?: { name?: string; description?: string };
  datacenter?: { name?: string; location?: { name?: string } };
  labels?: Record<string, string>;
};

type HetznerFirewall = {
  id: number;
  name: string;
  rules?: Array<{
    direction: string;
    protocol: string;
    port?: string;
    source_ips?: string[];
  }>;
  applied_to?: Array<{
    type: string;
    server?: { id: number };
  }>;
};

type HetznerSshKey = {
  id: number;
  name: string;
  public_key: string;
};

type InfraState = {
  version: 1;
  updatedAt: string;
  resources: {
    sshKeyId?: number;
    firewallId?: number;
    serverId?: number;
    serverIp?: string;
    serverName?: string;
  };
  lastRun?: {
    command: string;
    status: "ok" | "failed";
    at: string;
    note?: string;
  };
};

type Config = {
  token: string;
  serverName: string;
  firewallName: string;
  sshKeyName: string;
  serverType: string;
  serverImage: string;
  serverLocation: string;
  labelSelector: string;
  sshPrivateKeyPath: string;
  sshPublicKeyPath: string;
  statePath: string;

  expectedServerType: string;
};

function expandHome(p: string): string {
  return p.startsWith("~/") ? `${homedir()}/${p.slice(2)}` : p;
}

function shellQuote(value: string): string {
  return `'${value.replace(/'/g, `'\"'\"'`)}'`;
}

function run(command: string): string {
  return execSync(command, {
    stdio: ["ignore", "pipe", "pipe"],
    encoding: "utf8",
  }).trim();
}

function loadDotEnvFile(path: string): void {
  if (!existsSync(path)) return;
  const content = readFileSync(path, "utf8");
  for (const rawLine of content.split("\n")) {
    const line = rawLine.trim();
    if (!line || line.startsWith("#")) continue;
    const idx = line.indexOf("=");
    if (idx <= 0) continue;
    const key = line.slice(0, idx).trim();
    let value = line.slice(idx + 1).trim();
    if (
      (value.startsWith('"') && value.endsWith('"')) ||
      (value.startsWith("'") && value.endsWith("'"))
    ) {
      value = value.slice(1, -1);
    }
    if (!process.env[key]) {
      process.env[key] = value;
    }
  }
}

async function api<T>(config: Config, path: string, opts?: RequestInit): Promise<T> {
  const res = await fetch(`${API}${path}`, {
    ...opts,
    headers: {
      Authorization: `Bearer ${config.token}`,
      "Content-Type": "application/json",
      ...(opts?.headers || {}),
    },
  });
  if (!res.ok) {
    const body = await res.text();
    throw new Error(`Hetzner API ${res.status}: ${body}`);
  }
  return (await res.json()) as T;
}

function defaultState(): InfraState {
  return {
    version: 1,
    updatedAt: new Date().toISOString(),
    resources: {},
  };
}

function loadState(config: Config): InfraState {
  if (!existsSync(config.statePath)) return defaultState();
  try {
    const parsed = JSON.parse(readFileSync(config.statePath, "utf8")) as InfraState;
    if (parsed.version !== 1 || !parsed.resources) return defaultState();
    return parsed;
  } catch {
    return defaultState();
  }
}

function saveState(config: Config, state: InfraState): void {
  mkdirSync(dirname(config.statePath), { recursive: true });
  state.updatedAt = new Date().toISOString();
  writeFileSync(config.statePath, `${JSON.stringify(state, null, 2)}\n`, "utf8");
}

function markRun(
  config: Config,
  state: InfraState,
  command: string,
  status: "ok" | "failed",
  note?: string,
): void {
  state.lastRun = {
    command,
    status,
    at: new Date().toISOString(),
    note,
  };
  saveState(config, state);
}

function parseArgs(argv: string[]): {
  command: string;
  flags: Map<string, string>;
  boolFlags: Set<string>;
} {
  const [command = "status", ...rest] = argv;
  const flags = new Map<string, string>();
  const boolFlags = new Set<string>();
  for (const item of rest) {
    if (!item.startsWith("--")) continue;
    const eq = item.indexOf("=");
    if (eq === -1) {
      boolFlags.add(item.slice(2));
    } else {
      flags.set(item.slice(2, eq), item.slice(eq + 1));
    }
  }
  return { command, flags, boolFlags };
}

async function findServer(config: Config): Promise<HetznerServer | null> {
  const byLabel = await api<{ servers: HetznerServer[] }>(
    config,
    `/servers?label_selector=${encodeURIComponent(config.labelSelector)}`,
  );
  if (byLabel.servers.length) {
    const sorted = [...byLabel.servers].sort((a, b) => b.created.localeCompare(a.created));
    const running = sorted.find((s) => s.status === "running");
    return running || sorted[0]!;
  }

  const byName = await api<{ servers: HetznerServer[] }>(
    config,
    `/servers?name=${encodeURIComponent(config.serverName)}`,
  );
  if (byName.servers.length) return byName.servers[0]!;
  return null;
}

async function findFirewall(config: Config): Promise<HetznerFirewall | null> {
  const byName = await api<{ firewalls: HetznerFirewall[] }>(
    config,
    `/firewalls?name=${encodeURIComponent(config.firewallName)}`,
  );
  if (byName.firewalls.length) return byName.firewalls[0]!;
  return null;
}

async function ensureSshKey(config: Config, publicKey: string): Promise<HetznerSshKey> {
  const normalized = publicKey.trim();

  // First, match by exact public key to avoid uniqueness errors when
  // the key exists under a different name.
  const all = await api<{ ssh_keys: HetznerSshKey[] }>(config, "/ssh_keys");
  const byKey = all.ssh_keys.find((k) => k.public_key.trim() === normalized);
  if (byKey) return byKey;

  // Fallback: named lookup.
  const existing = await api<{ ssh_keys: HetznerSshKey[] }>(
    config,
    `/ssh_keys?name=${encodeURIComponent(config.sshKeyName)}`,
  );
  if (existing.ssh_keys.length) return existing.ssh_keys[0]!;

  const created = await api<{ ssh_key: HetznerSshKey }>(config, "/ssh_keys", {
    method: "POST",
    body: JSON.stringify({
      name: config.sshKeyName,
      public_key: normalized,
      labels: { project: "recall-protocol" },
    }),
  });
  return created.ssh_key;
}

async function ensureFirewall(config: Config): Promise<HetznerFirewall> {
  const existing = await findFirewall(config);
  if (existing) {
    const hasSshRule = (existing.rules || []).some(
      (rule) => rule.direction === "in" && rule.protocol === "tcp" && rule.port === "22",
    );
    if (!hasSshRule) {
      await api(config, `/firewalls/${existing.id}/actions/set_rules`, {
        method: "POST",
        body: JSON.stringify({
          rules: [
            {
              direction: "in",
              protocol: "tcp",
              port: "22",
              source_ips: ["0.0.0.0/0", "::/0"],
              description: "SSH bootstrap; remove after tunnel verification",
            },
          ],
        }),
      });
    }
    return existing;
  }

  const created = await api<{ firewall: HetznerFirewall }>(config, "/firewalls", {
    method: "POST",
    body: JSON.stringify({
      name: config.firewallName,
      labels: { project: "recall-protocol" },
      rules: [
        {
          direction: "in",
          protocol: "tcp",
          port: "22",
          source_ips: ["0.0.0.0/0", "::/0"],
          description: "SSH bootstrap; remove after tunnel verification",
        },
      ],
    }),
  });
  return created.firewall;
}

async function ensureServer(config: Config, sshKeyId: number, firewallId: number): Promise<HetznerServer> {
  const existing = await findServer(config);
  if (existing) return existing;

  const cloudInit = String.raw`#!/bin/bash
set -euo pipefail
exec > /var/log/cloud-init-script.log 2>&1
apt-get update && apt-get upgrade -y
curl -fsSL https://get.docker.com | sh
systemctl enable docker && systemctl start docker
apt-get install -y docker-compose-plugin
mkdir -p /opt/recall-protocol/mail-api /opt/recall-protocol/renderer /opt/recall-protocol/tls /opt/recall-protocol/dkim
touch /opt/recall-protocol/.cloud-init-done
echo "Cloud-init complete at $(date)"
`;

  const created = await api<{ server: HetznerServer }>(config, "/servers", {
    method: "POST",
    body: JSON.stringify({
      name: config.serverName,
      server_type: config.serverType,
      image: config.serverImage,
      location: config.serverLocation,
      ssh_keys: [sshKeyId],
      firewall_ids: [firewallId],
      user_data: cloudInit,
      labels: { project: "recall-protocol", environment: "production" },
    }),
  });
  return created.server;
}

async function ensureServerAttachedToFirewall(
  config: Config,
  firewallId: number,
  serverId: number,
): Promise<void> {
  const detail = await api<{ firewall: HetznerFirewall }>(config, `/firewalls/${firewallId}`);
  const attached = (detail.firewall.applied_to || []).some(
    (resource) => resource.type === "server" && resource.server?.id === serverId,
  );
  if (attached) return;

  // Best-effort attach: Hetzner has used multiple action routes across API surfaces.
  // We try common variants and only warn if none succeed.
  try {
    await api(config, `/servers/${serverId}/actions/attach_to_firewall`, {
      method: "POST",
      body: JSON.stringify({ firewall: firewallId }),
    });
    return;
  } catch {
    // Try firewall-side action route next.
  }

  try {
    await api(config, `/firewalls/${firewallId}/actions/apply_to_resources`, {
      method: "POST",
      body: JSON.stringify({
        apply_to: [{ type: "server", server: { id: serverId } }],
      }),
    });
    return;
  } catch (error) {
    console.warn(
      `[provision] warning: could not auto-attach firewall ${firewallId} to server ${serverId}: ${
        error instanceof Error ? error.message : String(error)
      }`,
    );
  }
}

function waitForSsh(host: string, privateKeyPath: string): void {
  for (let i = 1; i <= 60; i += 1) {
    try {
      run(
        `ssh -i ${shellQuote(privateKeyPath)} -o ConnectTimeout=5 -o StrictHostKeyChecking=accept-new root@${host} ${shellQuote(
          "test -f /opt/recall-protocol/.cloud-init-done && echo ready",
        )}`,
      );
      return;
    } catch {
      execSync("sleep 5");
    }
  }
  throw new Error("Timed out waiting for cloud-init / SSH readiness");
}

async function screenshotPng({ url, width, height }) {
  const browser = await chromium.launch({ headless: true, args: ["--no-sandbox", "--disable-setuid-sandbox"] });
  try {
    const page = await browser.newPage();
    await page.setViewportSize({ width: Number(width), height: Number(height) });
    await page.goto(url, { waitUntil: "networkidle", timeout: TIMEOUT_MS });
    return await page.screenshot({ type: "png" });
  } finally {
    await browser.close();
  }
}

function printServerStatus(server: HetznerServer | null): void {
  if (!server) {
    console.log("Server not found");
    return;
  }
  console.log(`ID:       ${server.id}`);
  console.log(`Server:   ${server.name}`);
  console.log(`Status:   ${server.status}`);
  console.log(`IP:       ${server.public_net.ipv4.ip}`);
  console.log(
    `Type:     ${server.server_type?.name || "unknown"} (${server.server_type?.description || "n/a"})`,
  );
  console.log(`Location: ${server.datacenter?.name || "unknown"}`);
  console.log(`Created:  ${server.created}`);
}

async function cmdProvision(config: Config, state: InfraState): Promise<void> {
  if (!config.token) throw new Error("HCLOUD_TOKEN is required");

  const sshPublicKey = readFileSync(config.sshPublicKeyPath, "utf8").trim();

  console.log(`[provision] ensuring ssh key: ${config.sshKeyName}`);
  const sshKey = await ensureSshKey(config, sshPublicKey);
  state.resources.sshKeyId = sshKey.id;

  console.log(`[provision] ensuring firewall: ${config.firewallName}`);
  const firewall = await ensureFirewall(config);
  state.resources.firewallId = firewall.id;

  console.log(`[provision] ensuring server: ${config.serverName}`);
  const server = await ensureServer(config, sshKey.id, firewall.id);
  state.resources.serverId = server.id;
  state.resources.serverIp = server.public_net.ipv4.ip;
  state.resources.serverName = server.name;
  await ensureServerAttachedToFirewall(config, firewall.id, server.id);
  saveState(config, state);

  console.log(
    `[provision] server id=${server.id} ip=${server.public_net.ipv4.ip} status=${server.status}`,
  );
  console.log("[provision] waiting for cloud-init + SSH...");
  waitForSsh(server.public_net.ipv4.ip, config.sshPrivateKeyPath);
  console.log("[provision] deploying remote config files...");
  writeRemoteConfig(server.public_net.ipv4.ip, config.sshPrivateKeyPath);

  console.log("[provision] complete");
}

async function cmdBootstrapTunnel(
  config: Config,
  state: InfraState,
  flags: Map<string, string>,
): Promise<void> {
  const tunnelToken =
    flags.get("token") ||
    process.env.TUNNEL_TOKEN ||
    process.env.CLOUDFLARE_TUNNEL_TOKEN ||
    "";
  if (!tunnelToken) {
    throw new Error("Missing tunnel token. Pass --token=<token> or set TUNNEL_TOKEN in infra/.env.");
  }

  const host = flags.get("host") || state.resources.serverIp || (await findServer(config))?.public_net.ipv4.ip;
  if (!host) {
    throw new Error("Could not determine host. Pass --host=<ip> or run `infra.ts provision` first.");
  }

  console.log(`[bootstrap-tunnel] target host: ${host}`);
  const remoteScript = `
set -euo pipefail
cd /opt/recall-protocol
cat > .env <<'EOF'
TUNNEL_TOKEN=${tunnelToken}
EOF
chmod 600 .env
docker compose up -d
docker compose ps
`;
  run(
    `ssh -i ${shellQuote(config.sshPrivateKeyPath)} -o StrictHostKeyChecking=accept-new root@${host} ${shellQuote(remoteScript)}`,
  );

  const health = run("curl -sS https://mail.recallprotocol.com/health");
  if (health !== "ok") {
    throw new Error(`Unexpected health response: ${health}`);
  }
  console.log("[bootstrap-tunnel] health check passed");
}

function checkSsh(host: string, sshKeyPath: string, failures: string[], warnings: string[]): void {
  const expectedPaths = [
    "/opt/recall-protocol/docker-compose.yml",
    "/opt/recall-protocol/Caddyfile",
    "/opt/recall-protocol/maddy.conf",
    "/opt/recall-protocol/mail-api/server.js",
    "/opt/recall-protocol/renderer/server.js",
    "/opt/recall-protocol/.secrets",
    "/opt/recall-protocol/tls/cert.pem",
    "/opt/recall-protocol/tls/key.pem",
    "/opt/recall-protocol/dkim/recallprotocol_default.pub.b64",
    "/opt/recall-protocol/dkim/recallprotocol_default.key",
  ];

  try {
    const remoteScript = `
set -euo pipefail
for p in ${expectedPaths.map((p) => shellQuote(p)).join(" ")}; do
  if [ ! -f "$p" ]; then
    echo "missing:$p"
  fi
done
if docker compose -f /opt/recall-protocol/docker-compose.yml config -q; then
  echo "compose:ok"
else
  echo "compose:invalid"
fi
if [ -f /opt/recall-protocol/.env ]; then
  echo "tunnel_env:present"
else
  echo "tunnel_env:missing"
fi
`;
    const out = run(
      `ssh -i ${shellQuote(sshKeyPath)} -o StrictHostKeyChecking=accept-new -o ConnectTimeout=10 root@${host} ${shellQuote(remoteScript)}`,
    );

    const lines = out.split("\n").map((line) => line.trim()).filter(Boolean);
    const missing = lines.filter((line) => line.startsWith("missing:"));
    if (missing.length) {
      for (const item of missing) {
        const message = `remote file missing: ${item.slice("missing:".length)}`;
        failures.push(message);
        console.log(`FAIL ${message}`);
      }
    } else {
      console.log("PASS all expected remote provisioned files exist");
    }
    if (!lines.includes("compose:ok")) {
      failures.push("docker compose config is invalid");
      console.log("FAIL docker compose config is invalid");
    } else {
      console.log("PASS docker compose config is valid");
    }
    if (!lines.includes("tunnel_env:present")) {
      const message = "tunnel runtime env missing (.env); run bootstrap-tunnel after tunnel token creation";
      warnings.push(message);
      console.log(`WARN ${message}`);
    } else {
      console.log("PASS tunnel runtime env file exists (.env)");
    }
  } catch (error) {
    const message = `ssh checks failed (${error instanceof Error ? error.message : String(error)})`;
    failures.push(message);
    console.log(`FAIL ${message}`);
  }
}

async function cmdCheck(
  config: Config,
  state: InfraState,
  boolFlags: Set<string>,
): Promise<void> {
  const apiOnly = boolFlags.has("api-only");
  const withSsh = boolFlags.has("with-ssh");
  if (apiOnly && withSsh) throw new Error("Use either --api-only or --with-ssh.");

  const failures: string[] = [];
  const warnings: string[] = [];
  const pass = (message: string) => console.log(`PASS ${message}`);
  const warn = (message: string) => {
    warnings.push(message);
    console.log(`WARN ${message}`);
  };
  const fail = (message: string) => {
    failures.push(message);
    console.log(`FAIL ${message}`);
  };

  const doSsh = withSsh || (!apiOnly && Boolean(process.env.SSH_PRIVATE_KEY_PATH));
  console.log(`drift check mode: ${doSsh ? "api+ssh" : "api-only"}`);

  const server = await findServer(config);
  if (!server) {
    fail(`server not found (expected label ${config.labelSelector} or name ${config.serverName})`);
  } else {
    pass(`server found: ${server.name} (${server.id})`);
    state.resources.serverId = server.id;
    state.resources.serverIp = server.public_net.ipv4.ip;
    state.resources.serverName = server.name;
    if (server.status !== "running") fail(`server status is ${server.status} (expected running)`);
    else pass("server status is running");

    const serverType = server.server_type?.name;
    if (serverType && serverType !== config.expectedServerType) {
      warn(`server type drift: got ${serverType}, expected ${config.expectedServerType}`);
    } else {
      pass(`server type matches expected (${config.expectedServerType})`);
    }

    const location = server.datacenter?.location?.name;
    if (location && location !== config.expectedLocation) {
      warn(`server location drift: got ${location}, expected ${config.expectedLocation}`);
    } else if (location) {
      pass(`server location matches expected (${config.expectedLocation})`);
    }

    if (server.labels?.project !== "recall-protocol") {
      warn("server label project=recall-protocol missing");
    } else {
      pass("server label project=recall-protocol is present");
    }
  }

  const firewall = await findFirewall(config);
  if (!firewall) {
    fail(`firewall not found (expected ${config.firewallName})`);
  } else {
    pass(`firewall found: ${firewall.name} (${firewall.id})`);
    state.resources.firewallId = firewall.id;
    const rules = firewall.rules || [];
    if (rules.length === 0) {
      pass("firewall has zero inbound rules (hardened)");
    } else {
      const allSsh = rules.every(
        (rule) => rule.direction === "in" && rule.protocol === "tcp" && rule.port === "22",
      );
      const allowedMailPorts = new Set(["22", "25", "465", "587"]);
      const allMailProfile = rules.every(
        (rule) =>
          rule.direction === "in" &&
          rule.protocol === "tcp" &&
          Boolean(rule.port) &&
          allowedMailPorts.has(String(rule.port)),
      );
      if (allSsh) pass("firewall is in bootstrap mode (only tcp/22 inbound)");
      else if (allMailProfile) pass("firewall is in mail mode (tcp/22,25,465,587)");
      else fail("firewall has unexpected inbound rules");
    }
  }

  if (doSsh) {
    if (!state.resources.serverIp) fail("cannot run ssh checks because no server IP is known");
    else checkSsh(state.resources.serverIp, config.sshPrivateKeyPath, failures, warnings);
  } else {
    warn("ssh checks skipped (use --with-ssh and set SSH_PRIVATE_KEY_PATH for deeper verification)");
  }

  saveState(config, state);
  console.log("");
  console.log("summary:");
  console.log(`  failures: ${failures.length}`);
  console.log(`  warnings: ${warnings.length}`);
  if (failures.length) throw new Error("drift check failed");
}

async function cmdStatus(config: Config, state: InfraState): Promise<void> {
  const server = await findServer(config);
  if (server) {
    state.resources.serverId = server.id;
    state.resources.serverIp = server.public_net.ipv4.ip;
    state.resources.serverName = server.name;
    saveState(config, state);
  }
  printServerStatus(server);
}

async function cmdFirewall(config: Config): Promise<void> {
  const fw = await findFirewall(config);
  if (!fw) {
    console.log("Firewall not found");
    return;
  }
  console.log(`Firewall: ${fw.name} (id: ${fw.id})`);
  if (!fw.rules || fw.rules.length === 0) {
    console.log("  No inbound rules (fully locked down)");
    return;
  }
  for (const rule of fw.rules) {
    const src = rule.source_ips?.join(", ") || "(none)";
    const port = rule.port || "(all)";
    console.log(`  ${rule.direction} ${rule.protocol}/${port} from ${src}`);
  }
}

async function cmdHarden(config: Config): Promise<void> {
  const fw = await findFirewall(config);
  if (!fw) {
    console.log("Firewall not found");
    return;
  }
  await api(config, `/firewalls/${fw.id}/actions/set_rules`, {
    method: "POST",
    body: JSON.stringify({ rules: [] }),
  });
  console.log("Firewall hardened: zero inbound rules");
}

async function cmdReboot(config: Config): Promise<void> {
  const server = await findServer(config);
  if (!server) {
    console.log("Server not found");
    return;
  }
  await api(config, `/servers/${server.id}/actions/reboot`, { method: "POST" });
  console.log(`Reboot initiated for ${server.name}`);
}

async function cmdIp(config: Config): Promise<void> {
  const server = await findServer(config);
  if (!server) {
    console.log("Server not found");
    return;
  }
  console.log(server.public_net.ipv4.ip);
}

function printUsage(): void {
  console.log("Usage: npm run infra -- <command> [flags]");
  console.log("");
  console.log("Commands:");
  console.log("  provision");
  console.log("  bootstrap-tunnel --token=<tunnel_token> [--host=<server_ip>]");
  console.log("  check [--api-only|--with-ssh]");
  console.log("  status | firewall | harden | reboot | ip");
}

async function main(): Promise<void> {
  const config = getConfig();
  const state = loadState(config);
  const parsed = parseArgs(process.argv.slice(2));

  if (parsed.command === "help" || parsed.command === "--help" || parsed.command === "-h") {
    printUsage();
    return;
  }

  if (!config.token) {
    throw new Error("HCLOUD_TOKEN missing. Set it in infra/.env or environment.");
  }

  try {
    switch (parsed.command) {
      case "provision":
        await cmdProvision(config, state);
        break;
      case "bootstrap-tunnel":
        await cmdBootstrapTunnel(config, state, parsed.flags);
        break;
      case "check":
        await cmdCheck(config, state, parsed.boolFlags);
        break;
      case "status":
        await cmdStatus(config, state);
        break;
      case "firewall":
        await cmdFirewall(config);
        break;
      case "harden":
        await cmdHarden(config);
        break;
      case "reboot":
        await cmdReboot(config);
        break;
      case "ip":
        await cmdIp(config);
        break;
      default:
        printUsage();
        throw new Error(`Unknown command: ${parsed.command}`);
    }
    markRun(config, state, parsed.command, "ok");
  } catch (error) {
    markRun(
      config,
      state,
      parsed.command,
      "failed",
      error instanceof Error ? error.message : String(error),
    );
    throw error;
  }
}

main().catch((error) => {
  console.error(error instanceof Error ? error.message : String(error));
  process.exit(1);
});
